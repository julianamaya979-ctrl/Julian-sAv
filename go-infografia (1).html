📏
Runes, Len & Conversión
Strings son bytes UTF-8 inmutables. Caracteres son runes.
// Longitud en bytes
n := len("Hola") // 4

// Acceder a byte por índice
c := "Hola"[0] // 72 (byte 'H')

// Iterar runes (unicode)
for i, r := range "héroe" {
  fmt.Printf("%d:%c ", i, r)
}

// Conversiones
b := []byte("hola") // []byte
r := []rune("héroe") // []rune
s := string(b) // back to string
runebyteUTF-8

Básicas & Multi-retorno
Declaración, sin/con retorno, múltiples retornos y retorno nombrado.
// Sin retorno (void)
func saludar(nombre string) {
  fmt.Println("Hola, " + nombre)
}

// Retorno simple
func sumar(a, b int) int {
  return a + b
}

// Multi-retorno (value, error)
func dividir(a, b float64) (float64, error) {
  if b == 0 { return 0, fmt.Errorf("÷0") }
  return a / b, nil
}

// Retorno nombrado
func minMax(nums []int) (min, max int) {
  min, max = nums[0], nums[0]
  return // naked return
}
multi-returnnamed return
🎯
Variádica, Anónima & Closure
Funciones de primer orden: variádicas, lambdas y closures.
// Variádica (...T)
func suma(nums ...int) int {
  t := 0
  for _, n := range nums { t += n }
  return t
}
suma(1,2,3) // 6
suma(slice...) // expand slice

// Función anónima (lambda)
doble := func(x int) int { return x * 2 }
doble(5) // 10

// Closure — captura variable externa
func contador() func() int {
  n := 0
  return func() int { n++; return n }
}
c := contador()
c() // 1
c() // 2
variadicclosurelambda

Métodos, init & Recursión
Métodos en structs, función especial init y funciones recursivas.
// Método sobre struct
type Rect struct { W, H float64 }

func (r Rect) Area() float64 {
  return r.W * r.H
}
// Receptor puntero (muta)
func (r *Rect) Scale(f float64) {
  r.W *= f; r.H *= f
}

// init — ejecuta al importar
func init() { fmt.Println("iniciado") }

// Recursión
func factorial(n int) int {
  if n <= 1 { return 1 }
  return n * factorial(n-1)
}
methodinitrecursion

Higher-Order Functions
Funciones que reciben o retornan otras funciones. Patrón funcional.
// Función como parámetro
func aplicar(nums []int, f func(int)int) []int {
  res := make([]int, len(nums))
  for i, v := range nums { res[i] = f(v) }
  return res
}

aplicar([]int{1,2,3}, func(x int) int {
  return x * x
}) // [1 4 9]

// Tipo función personalizado
type Transformar func(int) int
var op Transformar = func(x int) int {
  return x + 10
}

defer / panic / recover
Control de recursos y manejo de errores irrecuperables.
// defer — cleanup al salir
func leer(path string) {
  f, _ := os.Open(path)
  defer f.Close() // LIFO
}

// panic — error fatal
if db == nil { panic("db nil") }

// recover — captura panic
defer func() {
  if r := recover(); r != nil {
    fmt.Println("recuperado:", r)
  }
}()

⑦ Condicionales · if / else / switch / select
🔀
if / else if / else
Go permite inicialización en el if. Sin paréntesis en condición.
// Básico
if x > 0 {
  fmt.Println("positivo")
} else if x < 0 {
  fmt.Println("negativo")
} else {
  fmt.Println("cero")
}

// if con init (scope local)
if n := obtenerNum(); n > 10 {
  fmt.Println("grande:", n)
} else {
  fmt.Println("pequeño:", n)
}

// Error handling típico
if err != nil {
  return err
}
if initno parens
🎚️
switch
Sin fallthrough implícito. Soporta múltiples valores por case y type switch.
// Por valor
switch dia {
case "lunes":
  fmt.Println("inicio semana")
case "sábado", "domingo":
  fmt.Println("fin de semana")
default:
  fmt.Println("entre semana")
}

// Sin expresión (= if-else chain)
switch {
case x < 0: fmt.Println("negativo")
case x == 0: fmt.Println("cero")
default: fmt.Println("positivo")
}

// Type switch
switch v := i.(type) {
case int: fmt.Println("int", v)
case string: fmt.Println("str", v)
}

select & Operador ternario
select para channels. Go no tiene ?: pero sí patrones idiomáticos.
// select — espera channels
ch1 := make(chan string)
done := make(chan bool)

select {
case msg := <-ch1:
  fmt.Println("recibido:", msg)
case <-done:
  fmt.Println("terminado")
case <-time.After(1*time.Second):
  fmt.Println("timeout")
default:
  fmt.Println("sin bloqueo")
}

// "ternario" idiomático en Go
max := a
if b > a { max = b }

for Clásico
Equivalente al for de C/Java. La forma más explícita.
// for clásico: init; cond; post
for i := 0; i < 5; i++ {
  fmt.Println(i) // 0 1 2 3 4
}

// Cuenta regresiva
for i := 10; i > 0; i -= 2 {
  fmt.Print(i, " ") // 10 8 6 4 2
}

// Dos variables
for i, j := 0, 9; i < j; i++, j-- {
  fmt.Println(i, j)
}
classicmulti-var
⏳
for como While
Go no tiene while. Usa for condición como equivalente directo.
// while(x > 0)
x := 5
for x > 0 {
  fmt.Print(x, " ")
  x--
}
// Output: 5 4 3 2 1

// do-while simulado
for {
  fmt.Scan(&x)
  if x == 0 { break }
}

// Flag booleano
corriendo := true
for corriendo {
  corriendo = procesarEvento()
}

♾️
Loop Infinito & Control
for {} es infinito. break, continue y labels para control de flujo.
// Loop infinito — siempre necesita break
for {
  cmd := leerComando()
  if cmd == "exit" { break }
  if cmd == "" { continue }
  ejecutar(cmd)
}

// Labeled break (loops anidados)
Externo:
for i := 0; i < 3; i++ {
  for j := 0; j < 3; j++ {
    if i+j == 3 {
      break Externo // sale de ambos
    }
  }
}

Patrones Funcionales con Loops
Reduce, filter, map, buscar y acumular con bucles idiomáticos de Go.
nums := []int{1,2,3,4,5,6}

// Reduce (sumar)
total := 0
for _, v := range nums { total += v }

// Filter (solo pares)
pares := []int{}
for _, v := range nums {
  if v%2==0 { pares = append(pares,v) }
}

// Map transform (cuadrados)
cuad := make([]int, len(nums))
for i, v := range nums { cuad[i]=v*v }

// Find (buscar primero)
for _, v := range nums {
  if v > 3 { fmt.Println("found:",v); break }
